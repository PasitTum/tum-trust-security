### Type A — OAuth 2.0 / OpenID Connect (ต้นแบบจาก `IThaIdProvider`)

```csharp
using Tum.Trust.Security.Core.Models;

namespace Tum.Trust.Security.Core.Interfaces;

/// <summary>
/// Defines the contract for <Name> (OAuth 2.0 / OIDC) operations.
/// Each method corresponds directly to an official <Name> endpoint and returns the provider response unmodified.
/// </summary>
public interface I<Name>Provider : IVerificationProvider
{
    /// <summary>
    /// Generates the <Name> OAuth 2.0 authorization URL (with optional PKCE and OIDC nonce parameters).
    /// </summary>
    /// <param name="redirectUri">Callback URI registered with <Name>.</param>
    /// <param name="state">CSRF state token generated by the caller.</param>
    /// <param name="codeChallenge">Optional PKCE code challenge (S256 Base64Url string).</param>
    /// <param name="codeChallengeMethod">PKCE challenge method (defaults to "S256").</param>
    /// <param name="nonce">Optional OIDC nonce parameter.</param>
    /// <returns>The full <Name> authorization URL.</returns>
    string GetAuthorizationUrl(
        string redirectUri,
        string state,
        string? codeChallenge,
        string codeChallengeMethod = "S256",
        string? nonce = null);

    /// <summary>
    /// Exchanges an OAuth 2.0 authorization code for tokens at the <Name> Token Endpoint.
    /// </summary>
    /// <param name="authorizationCode">The authorization code received from the callback.</param>
    /// <param name="redirectUri">Optional redirect URI (uses configured RedirectUri if omitted).</param>
    /// <param name="codeVerifier">Optional PKCE code verifier if PKCE was used during authorization.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns>A <see cref="VerificationResult"/> containing the raw token response.</returns>
    Task<VerificationResult> ExchangeCodeAsync(
        string authorizationCode,
        string? redirectUri = null,
        string? codeVerifier = null,
        CancellationToken cancellationToken = default);

    /// <summary>
    /// Requests a new access token using a refresh token.
    /// </summary>
    /// <param name="refreshToken">The refresh token previously issued by <Name>.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns>A <see cref="VerificationResult"/> containing the new token response.</returns>
    Task<VerificationResult> RefreshTokenAsync(
        string refreshToken,
        CancellationToken cancellationToken = default);

    /// <summary>
    /// Retrieves user profile attributes from the <Name> UserInfo Endpoint.
    /// </summary>
    /// <param name="accessToken">The valid Bearer access token.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns>A <see cref="VerificationResult"/> containing the user profile attributes.</returns>
    Task<VerificationResult> GetUserInfoAsync(
        string accessToken,
        CancellationToken cancellationToken = default);
}
```

## Template 2: Provider Class (`Infrastructure/Providers/<Name>Provider.cs`)

### Type A Implementation Skeleton

```csharp
using System.Net.Http.Headers;
using System.Text;
using Tum.Trust.Security.Core.Attributes;
using Tum.Trust.Security.Core.Exceptions;
using Tum.Trust.Security.Core.Interfaces;
using Tum.Trust.Security.Core.Models;
using Tum.Trust.Security.Infrastructure.Helpers;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Logging;

namespace Tum.Trust.Security.Infrastructure.Providers;

/// <summary>
/// <Name> identity verification provider implementing <see cref="I<Name>Provider"/>.
/// </summary>
[AuditLog("IdentityVerification", "<Name>", Description = "Verify identity with <Name>")]
public class <Name>Provider : I<Name>Provider
{
    /// <inheritdoc />
    public string ProviderName => "<Name>";

    private readonly string _clientId;
    private readonly string _clientSecret;
    private readonly string _authorizationUrl;
    private readonly string _tokenUrl;
    private readonly string? _userInfoUrl;
    private readonly string? _defaultRedirectUri;
    private readonly string _scope;

    private readonly HttpClient _httpClient;
    private readonly IEncryptionService _encryptionService;
    private readonly ILogger<<Name>Provider> _logger;

    /// <summary>
    /// Initializes a new instance of <see cref="<Name>Provider"/>.
    /// </summary>
    public <Name>Provider(
        IConfiguration configuration,
        HttpClient httpClient,
        IEncryptionService encryptionService,
        ILogger<<Name>Provider> logger)
    {
        if (configuration is null) throw new ArgumentNullException(nameof(configuration));
        if (httpClient is null) throw new ArgumentNullException(nameof(httpClient));
        if (encryptionService is null) throw new ArgumentNullException(nameof(encryptionService));
        if (logger is null) throw new ArgumentNullException(nameof(logger));

        var section = configuration.GetSection("IdentityVerification:<Name>");

        _clientId = section.GetRequiredValue("ClientId");
        _clientSecret = section.GetRequiredValue("ClientSecret");
        _authorizationUrl = section.GetRequiredValue("AuthorizationUrl");
        _tokenUrl = section.GetRequiredValue("TokenUrl");
        _userInfoUrl = section["UserInfoUrl"];
        _defaultRedirectUri = section["RedirectUri"];
        _scope = !string.IsNullOrWhiteSpace(section["Scope"])
            ? section["Scope"]!
            : "openid profile email";

        _httpClient = httpClient;
        _encryptionService = encryptionService;
        _logger = logger;
    }

    /// <inheritdoc />
    public string GetAuthorizationUrl(string redirectUri, string state)
    {
        return GetAuthorizationUrl(redirectUri, state, codeChallenge: null);
    }

    /// <inheritdoc />
    public string GetAuthorizationUrl(
        string redirectUri,
        string state,
        string? codeChallenge,
        string codeChallengeMethod = "S256",
        string? nonce = null)
    {
        var targetRedirectUri = !string.IsNullOrWhiteSpace(redirectUri)
            ? redirectUri
            : _defaultRedirectUri;

        if (string.IsNullOrWhiteSpace(targetRedirectUri))
            throw new ArgumentException("Redirect URI cannot be null or whitespace.", nameof(redirectUri));

        if (string.IsNullOrWhiteSpace(state))
            throw new ArgumentException("State cannot be null or whitespace.", nameof(state));

        var query = new Dictionary<string, string>
        {
            ["client_id"] = _clientId,
            ["redirect_uri"] = targetRedirectUri,
            ["response_type"] = "code",
            ["scope"] = _scope,
            ["state"] = state
        };

        if (!string.IsNullOrWhiteSpace(nonce))
        {
            query["nonce"] = nonce;
        }

        if (!string.IsNullOrWhiteSpace(codeChallenge))
        {
            query["code_challenge"] = codeChallenge;
            query["code_challenge_method"] = string.IsNullOrWhiteSpace(codeChallengeMethod) ? "S256" : codeChallengeMethod;
        }

        var queryString = string.Join("&", query.Select(kvp =>
            $"{Uri.EscapeDataString(kvp.Key)}={Uri.EscapeDataString(kvp.Value)}"));

        return $"{_authorizationUrl}?{queryString}";
    }

    /// <inheritdoc />
    [AuditLog("VerifyAsync", "<Name>", TrackDiff = false)]
    public async Task<VerificationResult> VerifyAsync(
        VerificationRequest request,
        CancellationToken cancellationToken = default)
    {
        if (request is null) throw new ArgumentNullException(nameof(request));

        var expectedState = request.AdditionalParameters.GetStringValue("expected_state");
        if (!string.IsNullOrEmpty(expectedState) &&
            !string.Equals(request.State, expectedState, StringComparison.Ordinal))
        {
            return ProviderHelper.CreateFailureResult("INVALID_STATE", "OAuth state validation failed.");
        }

        var codeVerifier = request.AdditionalParameters.GetStringValue("code_verifier");
        return await ExchangeCodeAsync(
            request.AuthorizationCode ?? string.Empty,
            request.RedirectUri,
            codeVerifier,
            cancellationToken);
    }

    /// <inheritdoc />
    [AuditLog("ExchangeCodeAsync", "<Name>", TrackDiff = false)]
    public async Task<VerificationResult> ExchangeCodeAsync(
        string authorizationCode,
        string? redirectUri = null,
        string? codeVerifier = null,
        CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrWhiteSpace(authorizationCode))
        {
            return ProviderHelper.CreateFailureResult("MISSING_AUTH_CODE", "Authorization code is required.");
        }

        var targetRedirectUri = !string.IsNullOrWhiteSpace(redirectUri)
            ? redirectUri
            : _defaultRedirectUri;

        if (string.IsNullOrWhiteSpace(targetRedirectUri))
        {
            return ProviderHelper.CreateFailureResult("MISSING_REDIRECT_URI", "Redirect URI is required.");
        }

        var formFields = new Dictionary<string, string>
        {
            ["grant_type"] = "authorization_code",
            ["code"] = authorizationCode,
            ["client_id"] = _clientId,
            ["client_secret"] = _clientSecret,
            ["redirect_uri"] = targetRedirectUri
        };

        if (!string.IsNullOrWhiteSpace(codeVerifier))
        {
            formFields["code_verifier"] = codeVerifier;
        }

        return await ExecutePostFormAsync(
            _tokenUrl,
            formFields,
            "TOKEN_EXCHANGE_FAILED",
            "<Name> token exchange successful",
            cancellationToken);
    }

    /// <inheritdoc />
    [AuditLog("RefreshTokenAsync", "<Name>", TrackDiff = false)]
    public async Task<VerificationResult> RefreshTokenAsync(
        string refreshToken,
        CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrWhiteSpace(refreshToken))
        {
            return ProviderHelper.CreateFailureResult("MISSING_REFRESH_TOKEN", "Refresh token is required.");
        }

        var formFields = new Dictionary<string, string>
        {
            ["grant_type"] = "refresh_token",
            ["refresh_token"] = refreshToken
        };

        return await ExecutePostFormAsync(
            _tokenUrl,
            formFields,
            "TOKEN_REFRESH_FAILED",
            "<Name> token refresh successful",
            cancellationToken);
    }

    /// <inheritdoc />
    [AuditLog("GetUserInfoAsync", "<Name>", TrackDiff = false)]
    public async Task<VerificationResult> GetUserInfoAsync(
        string accessToken,
        CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrWhiteSpace(_userInfoUrl))
        {
            return ProviderHelper.CreateFailureResult(
                "MISSING_USERINFO_URL",
                "IdentityVerification:<Name>:UserInfoUrl is not configured.");
        }

        if (string.IsNullOrWhiteSpace(accessToken))
        {
            return ProviderHelper.CreateFailureResult("MISSING_ACCESS_TOKEN", "Access token is required.");
        }

        try
        {
            using var httpRequest = new HttpRequestMessage(HttpMethod.Get, _userInfoUrl);
            httpRequest.Headers.Authorization = new AuthenticationHeaderValue("Bearer", accessToken);

            using var response = await _httpClient.SendAsync(httpRequest, cancellationToken);
            if (!response.IsSuccessStatusCode)
            {
                return ProviderHelper.CreateFailureResult(
                    "USERINFO_FETCH_FAILED",
                    $"<Name> userinfo request failed with HTTP {(int)response.StatusCode}.",
                    (int)response.StatusCode);
            }

            var json = await response.Content.ReadAsStringAsync(cancellationToken);
            var responseData = ProviderHelper.ParseJsonToDictionary(json, "INVALID_USERINFO_RESPONSE");

            _logger.LogEncryptedPiiIfPresent(_encryptionService, ProviderName, responseData, "pid", "citizen_id", "sub", "email");

            return new VerificationResult
            {
                IsSuccess = true,
                Message = "<Name> userinfo retrieval successful",
                Data = responseData
            };
        }
        catch (VerificationException ex)
        {
            _logger.LogError(ex, "<Name> GetUserInfoAsync failed: {ErrorCode}", ex.ErrorCode);
            return ProviderHelper.CreateFailureResult(ex.ErrorCode, ex.Message, ex.HttpStatusCode);
        }
        catch (HttpRequestException ex)
        {
            _logger.LogError(ex, "Network error calling <Name> userinfo endpoint.");
            return ProviderHelper.CreateFailureResult("NETWORK_ERROR", $"Network error: {ex.Message}");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Unexpected error in <Name> GetUserInfoAsync.");
            return ProviderHelper.CreateFailureResult("INTERNAL_ERROR", "An unexpected error occurred.");
        }
    }

    private async Task<VerificationResult> ExecutePostFormAsync(
        string endpointUrl,
        IDictionary<string, string> formFields,
        string errorCodeOnFailure,
        string successMessage,
        CancellationToken cancellationToken)
    {
        try
        {
            using var httpRequest = new HttpRequestMessage(HttpMethod.Post, endpointUrl)
            {
                Content = new FormUrlEncodedContent(
                    formFields.Select(kvp => new KeyValuePair<string?, string?>(kvp.Key, kvp.Value)))
            };

            var basicCredentials = Convert.ToBase64String(Encoding.UTF8.GetBytes($"{_clientId}:{_clientSecret}"));
            httpRequest.Headers.Authorization = new AuthenticationHeaderValue("Basic", basicCredentials);

            using var response = await _httpClient.SendAsync(httpRequest, cancellationToken);
            if (!response.IsSuccessStatusCode)
            {
                return ProviderHelper.CreateFailureResult(
                    errorCodeOnFailure,
                    $"<Name> request failed with HTTP {(int)response.StatusCode}.",
                    (int)response.StatusCode);
            }

            var json = await response.Content.ReadAsStringAsync(cancellationToken);
            var responseData = ProviderHelper.ParseJsonToDictionary(json, "INVALID_<NAME>_RESPONSE");

            _logger.LogEncryptedPiiIfPresent(_encryptionService, ProviderName, responseData, "pid", "citizen_id", "sub", "email");

            return new VerificationResult
            {
                IsSuccess = true,
                Message = successMessage,
                Data = responseData
            };
        }
        catch (VerificationException ex)
        {
            _logger.LogError(ex, "<Name> operation failed: {ErrorCode}", ex.ErrorCode);
            return ProviderHelper.CreateFailureResult(ex.ErrorCode, ex.Message, ex.HttpStatusCode);
        }
        catch (HttpRequestException ex)
        {
            _logger.LogError(ex, "Network error calling <Name> endpoint {EndpointUrl}.", endpointUrl);
            return ProviderHelper.CreateFailureResult("NETWORK_ERROR", $"Network error: {ex.Message}");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Unexpected error calling <Name> endpoint {EndpointUrl}.", endpointUrl);
            return ProviderHelper.CreateFailureResult("INTERNAL_ERROR", "An unexpected error occurred.");
        }
    }
}
```

## Template 3: Unit Tests (`Tests/<Name>ProviderTests.cs`)

```csharp
using System.Net;
using System.Security.Cryptography;
using Tum.Trust.Security.Core.Models;
using Tum.Trust.Security.Infrastructure.Providers;
using Tum.Trust.Security.Infrastructure.Security;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Logging.Abstractions;
using Xunit;

namespace Tum.Trust.Security.Tests;

public class <Name>ProviderTests
{
    private static IConfiguration CreateConfiguration()
    {
        var settings = new Dictionary<string, string?>
        {
            ["IdentityVerification:<Name>:ClientId"] = "test-client-id",
            ["IdentityVerification:<Name>:ClientSecret"] = "test-client-secret",
            ["IdentityVerification:<Name>:AuthorizationUrl"] = "https://provider.local/oauth/authorize",
            ["IdentityVerification:<Name>:TokenUrl"] = "https://provider.local/oauth/token",
            ["IdentityVerification:<Name>:UserInfoUrl"] = "https://provider.local/oauth/userinfo",
            ["IdentityVerification:<Name>:RedirectUri"] = "https://kiosk.local/callback",
            ["IdentityVerification:<Name>:Scope"] = "openid profile email"
        };

        return new ConfigurationBuilder()
            .AddInMemoryCollection(settings)
            .Build();
    }

    private static <Name>Provider CreateProvider(MockHttpMessageHandler handler)
    {
        byte[] keyBytes = new byte[32];
        RandomNumberGenerator.Fill(keyBytes);
        string key = Convert.ToBase64String(keyBytes);

        return new <Name>Provider(
            CreateConfiguration(),
            new HttpClient(handler),
            new AesEncryptionService(key),
            NullLogger<<Name>Provider>.Instance);
    }

    [Fact]
    public void GetAuthorizationUrl_IncludesPkceAndStateParameters()
    {
        var handler = new MockHttpMessageHandler(_ => MockHttpMessageHandler.JsonResponse("{}"));
        var provider = CreateProvider(handler);

        string url = provider.GetAuthorizationUrl(
            redirectUri: "https://kiosk.local/callback",
            state: "state-123",
            codeChallenge: "challenge-xyz",
            codeChallengeMethod: "S256",
            nonce: "nonce-999");

        Assert.Contains("response_type=code", url);
        Assert.Contains("client_id=test-client-id", url);
        Assert.Contains("state=state-123", url);
        Assert.Contains("code_challenge=challenge-xyz", url);
    }

    [Fact]
    public async Task ExchangeCodeAsync_ReturnsUnmodifiedTokenResponse()
    {
        string tokenJson = """
        {
            "access_token": "provider-access-token",
            "refresh_token": "provider-refresh-token",
            "token_type": "Bearer",
            "expires_in": 3600
        }
        """;

        var handler = new MockHttpMessageHandler(_ => MockHttpMessageHandler.JsonResponse(tokenJson));
        var provider = CreateProvider(handler);

        var result = await provider.ExchangeCodeAsync("auth-code-001", "https://kiosk.local/callback", "verifier-001");

        Assert.True(result.IsSuccess);
        Assert.Equal("provider-access-token", result.Data["access_token"]);
        Assert.Equal("provider-refresh-token", result.Data["refresh_token"]);
    }

    [Fact]
    public async Task ExchangeCodeAsync_WhenHttpError_ReturnsFailureResult()
    {
        var handler = new MockHttpMessageHandler(_ =>
            MockHttpMessageHandler.JsonResponse("{\"error\":\"invalid_grant\"}", HttpStatusCode.BadRequest));
        var provider = CreateProvider(handler);

        var result = await provider.ExchangeCodeAsync("bad-code", "https://kiosk.local/callback");

        Assert.False(result.IsSuccess);
        Assert.Equal("TOKEN_EXCHANGE_FAILED", result.ErrorCode);
        Assert.Equal(400, result.Error?.HttpStatusCode);
    }
}
```
